import java.util.Scanner;

public class Solution {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        if (in.hasNextLong()) {
            long a = in.nextLong();
            long b = in.nextLong();
            System.out.println(a + b);
        }
    }
}
